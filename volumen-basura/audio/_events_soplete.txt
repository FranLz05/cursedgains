def roar(t0, t1, g=0.5):          # a blowtorch roar: filtered noise with flutter
    d = t1 - t0; n = noise(d); x = bp(n, 180, 1800, 2) * (0.7 + 0.3 * np.sin(2 * np.pi * 9 * tt(d)))
